function enviarRegistro() { if (!ultimoCodigo) { alert('Primero debes escanear un boleto.'); return; } let enlace = localStorage.getItem('liamRegistroUrl'); if (!enlace) { enlace = prompt( 'Pega la URL /exec de tu aplicación privada de Apps Script:' ); if (!enlace) return; enlace = enlace.trim(); } try { const url = new URL(enlace); if ( url.protocol !== 'https:' || url.hostname !== 'script.google.com' || !url.pathname.endsWith('/exec') ) { alert('La dirección no corresponde a una aplicación web de Apps Script.'); return; } localStorage.setItem('liamRegistroUrl', url.origin + url.pathname); url.searchParams.set('qr', ultimoCodigo); const ventana = window.open(url.toString(), '_blank'); if (!ventana) { alert('Permite las ventanas emergentes para abrir el registro.'); } } catch (error) { alert('Revisa la dirección de Google Apps Script.'); } }
